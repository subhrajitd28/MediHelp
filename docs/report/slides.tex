% ----- 2. PROBLEM -----
\begin{frame}{What problem are we solving?}
  \begin{itemize}
    \item Existing health apps split into two camps:
      \begin{itemize}
        \item \textbf{Closed-source consumer apps} (Apple Health, Google Fit) --- proprietary, locked to one OS, no AI advice
        \item \textbf{Clinical EHRs} (OpenMRS, OpenEMR) --- doctor-facing, not patient-facing
      \end{itemize}
    \item Symptom checkers are clinically inaccurate (Semigran 2015 --- BMJ)
    \item LLMs hallucinate; pure-LLM medical advice is unsafe
    \item No tool combines: \emph{symptom triage + cultural-food advice + record export + family hub + emergency SOS} for the Indian patient
  \end{itemize}
  \vfill
  \begin{exampleblock}{Thesis goal}
    Build a multi-user web application that integrates these capabilities,
    proves they can be built with free-tier resources, and is architecturally
    sound enough to discuss in a B.Tech viva.
  \end{exampleblock}
\end{frame}

% ----- 3. APPROACH -----
\begin{frame}{Our approach}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Architectural} choices:
    \begin{itemize}
      \item 8 microservices (Java + Python), Eureka registry
      \item \alert{JWT validated only at gateway} --- downstream services read trusted \texttt{X-User-Id} header
      \item Two communication channels only: sync HTTP via gateway, async events via RabbitMQ
      \item One Postgres per service (DB isolation)
    \end{itemize}
    \column{0.45\textwidth}
    \textbf{AI safety} mechanisms:
    \begin{itemize}
      \item RAG over 5,775 chunks (Pinecone) grounds the LLM
      \item Pure-LLM fallback only if RAG returns ``general''
      \item \alert{3-layer severity classifier} --- worst-of-three
      \item Smalltalk gate to avoid LLM cost on greetings
      \item Drug-interaction safety net via OpenFDA
    \end{itemize}
  \end{columns}
\end{frame}

% ----- 5. RAG PIPELINE -----
\begin{frame}{The 5-stage chatbot pipeline}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{enumerate}\small
      \item \textbf{Symptom extraction} --- structured JSON from Llama-3.3
      \item \textbf{Disease RAG} --- Pinecone \texttt{general} k=4
      \item \textbf{Nutrition RAG} --- Pinecone \texttt{nutrition} k=5, 7-row table
      \item \textbf{Severity} --- 3-layer worst-of (rules + keywords + LLM)
      \item \textbf{Home care} --- skipped if CRITICAL
    \end{enumerate}
    \vspace{4pt}
    Gates around the pipeline:
    \begin{itemize}\small
      \item \alert{Smalltalk gate} --- no LLM cost for ``hi''
      \item \alert{Fallback gate} --- pure-LLM if RAG says ``general''
      \item \alert{Critical gate} --- red modal + tel:108 button
    \end{itemize}
    \column{0.45\textwidth}

  \end{columns}
\end{frame}

% ----- 7. AUTHENTICATION FLOW -----
\begin{frame}[fragile]{Auth: header-injection pattern}
  Gateway is the only service that parses JWTs. Downstream services trust three
  injected headers:
  \begin{lstlisting}[basicstyle=\ttfamily\small, frame=single, framerule=0.5pt]
@GetMapping("/me")
public ApiResponse<UserProfileDto> getMe(
    @RequestHeader("X-User-Id") String userId) {
  return ApiResponse.success(userService.getProfile(userId));
}
  \end{lstlisting}
  Benefits:
  \begin{itemize}
    \item Polyglot: Java + Python both read headers; no JWT library duplication
    \item Logout uses Redis blacklist on \texttt{jti}; downstream is unaffected
    \item Token refresh deduplicated in Angular interceptor (concurrent 401s share one refresh call)
  \end{itemize}
\end{frame}

% ----- 10. OBSERVABILITY -----
\begin{frame}{Self-hosted observability}
  \begin{columns}
    \column{0.55\textwidth}
    \begin{itemize}
      \item Prometheus scrapes \texttt{/actuator/prometheus} every 15s
      \item Grafana dashboard auto-provisioned from JSON
      \item Tracks: JVM heap, RPS, p95 latency, GC pauses, threads, services-up
    \end{itemize}
    \vspace{6pt}
    Observed in production:
    \begin{itemize}\small
      \item Heap comfortably under \texttt{-Xmx256m}
      \item p95 latency \textless 100~ms (non-LLM endpoints)
      \item Chatbot \texttt{/get}: 8--12~s (5--8 sequential LLM calls)
    \end{itemize}
